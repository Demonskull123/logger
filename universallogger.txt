local function isLoco(id, name)
	if LOCO[name] then return true end
	for _, f in ipairs(LOCO_FRAG) do
		if id:find(f) then return true end
	end
	return false
end

local function toRbxAsset(id)
	if type(id) ~= "string" then return id end
	local num = id:match("id=(%d+)") or id:match("rbxassetid://(%d+)") or id:match("^(%d+)$")
	if num then return "rbxassetid://" .. num end
	return id
end

local function toNum(id)
	if type(id) ~= "string" then return id end
	return id:match("id=(%d+)") or id:match("rbxassetid://(%d+)") or id:match("^(%d+)$") or id
end

local function copyId(id)
	local n = toNum(id)
	if setclipboard then
		pcall(setclipboard, tostring(n))
	elseif setrbxclipboard then
		pcall(setrbxclipboard, tostring(n))
	end
end

local timings = {}
local dots = {}

local function loadTimings()
	local ok, raw = pcall(readfile, FILE)
	if not ok or not raw or raw == "" then return end
	local ok2, tbl = pcall(function() return HttpService:JSONDecode(raw) end)
	if ok2 and type(tbl) == "table" then timings = tbl end
end

local function saveTimings()
	pcall(function() writefile(FILE, HttpService:JSONEncode(timings)) end)
end

local function refreshDots()
	for id, d in pairs(dots) do
		if d and d.Parent then
			d.BackgroundColor3 = timings[id]
				and Color3.fromRGB(80, 220, 120)
				or Color3.fromRGB(70, 70, 70)
		end
	end
end

local function mkGui(name)
	for _, r in ipairs({ gethui and gethui() or nil, game:GetService("CoreGui"), LP:FindFirstChild("PlayerGui") }) do
		if r then
			local old = r:FindFirstChild(name)
			if old then pcall(function() old:Destroy() end) end
		end
	end
	local g = Instance.new("ScreenGui")
	g.Name = name
	g.ResetOnSpawn = false
	g.ZIndexBehavior = Enum.ZIndexBehavior.Sibling
	pcall(function() g.Parent = gethui() end)
	if not g.Parent then pcall(function() g.Parent = game:GetService("CoreGui") end) end
	if not g.Parent then g.Parent = LP:WaitForChild("PlayerGui") end
	return g
end

-- ===== Movement lock =====
local MoveLock = {
	enabled = false,
	saved = nil,
}

local function applyMoveLock()
	local char = LP.Character
	if not char then return end
	local hum = char:FindFirstChildOfClass("Humanoid")
	if not hum then return end
	if not MoveLock.saved then
		MoveLock.saved = {
			WalkSpeed    = hum.WalkSpeed,
			JumpPower    = hum.JumpPower,
			JumpHeight   = hum.JumpHeight,
			UseJumpPower = hum.UseJumpPower,
		}
	end
	hum.WalkSpeed = 0
	if hum.UseJumpPower then
		hum.JumpPower = 0
	else
		hum.JumpHeight = 0
	end
	hum:SetStateEnabled(Enum.HumanoidStateType.Jumping, false)
	hum:SetStateEnabled(Enum.HumanoidStateType.Freefall, false)
	hum:SetStateEnabled(Enum.HumanoidStateType.Swimming, false)
	hum:SetStateEnabled(Enum.HumanoidStateType.Climbing, false)
	hum:SetStateEnabled(Enum.HumanoidStateType.FallingDown, false)
	hum:SetStateEnabled(Enum.HumanoidStateType.Ragdoll, false)
	hum:SetStateEnabled(Enum.HumanoidStateType.PlatformStanding, false)
	hum:ChangeState(Enum.HumanoidStateType.Seated)
	local jumpCtrl = hum:FindFirstChild("JumpControl") or hum:FindFirstChild("Controller")
	if jumpCtrl then pcall(function() jumpCtrl:Destroy() end) end
end

local function restoreMoveLock()
	local char = LP.Character
	if not char then return end
	local hum = char:FindFirstChildOfClass("Humanoid")
	if not hum then return end
	if MoveLock.saved then
		hum.WalkSpeed    = MoveLock.saved.WalkSpeed
		hum.JumpPower    = MoveLock.saved.JumpPower
		hum.JumpHeight   = MoveLock.saved.JumpHeight
		hum.UseJumpPower = MoveLock.saved.UseJumpPower
	end
	hum:SetStateEnabled(Enum.HumanoidStateType.Jumping, true)
	hum:SetStateEnabled(Enum.HumanoidStateType.Freefall, true)
	hum:SetStateEnabled(Enum.HumanoidStateType.Swimming, true)
	hum:SetStateEnabled(Enum.HumanoidStateType.Climbing, true)
	hum:SetStateEnabled(Enum.HumanoidStateType.FallingDown, true)
	hum:SetStateEnabled(Enum.HumanoidStateType.Ragdoll, true)
	hum:SetStateEnabled(Enum.HumanoidStateType.PlatformStanding, true)
end

local function setMoveLock(on)
	MoveLock.enabled = on
	if on then
		applyMoveLock()
	else
		restoreMoveLock()
	end
end

LP.CharacterAdded:Connect(function(c)
	local hum = c:WaitForChild("Humanoid", 5)
	if hum and MoveLock.enabled then
		task.wait(0.2)
		MoveLock.saved = nil
		applyMoveLock()
	end
end)

local Visualizer = {}

function Visualizer.selectAnim(id)
	if type(id) ~= "string" or id == "" then return end
	if Visualizer.frame then
		Visualizer.currentId = id
		if Visualizer.idBar then Visualizer.idBar.Text = toRbxAsset(id) end
		Visualizer.frame.Visible = true
		if Visualizer.playFn then Visualizer.playFn(id) end
	else
		Visualizer.pending = id
	end
end

			hum.AutoRotate = false
			hum.EvaluateStateMachine = false
			hum.RigType = hum0.RigType
			local old = hum:FindFirstChildOfClass("Animator")
			if old then old:Destroy() end
			local a = Instance.new("Animator")
			a.Parent = hum
			Visualizer.hum = hum
		end

		local hrp = model:FindFirstChild("HumanoidRootPart")
		if hrp then
			-- R15: HRP sits ~2.5 studs above the feet. Place feet on plate top (Y=0).
			model:PivotTo(CFrame.new(0, 2.5, 0))
			hrp.Anchored = true
		end

		Visualizer.rig = model
	end

	buildRig()

	local function stop()
		if Visualizer.track then
			pcall(function() Visualizer.track:Stop() end)
			Visualizer.track = nil
		end
		Visualizer.paused = false
		scrubFill.Size = UDim2.new(0, 0, 1, 0)
		timeLbl.Text = "0.000s / 0.000s"
		btnPause.Text = "Pause"
	end

	local function play(id)
		if not Visualizer.hum or not Visualizer.hum.Parent then buildRig() end
		if not Visualizer.hum then return end

		stop()

		local useId = toRbxAsset(id)

		local a = Instance.new("Animation")
		a.AnimationId = useId

		local okL, track = pcall(function() return Visualizer.hum:LoadAnimation(a) end)
		if not okL or not track then return end

		track.Looped = Visualizer.looping
		track:Play()
		pcall(function() track:AdjustSpeed(Visualizer.speed) end)

		Visualizer.track = track
		Visualizer.duration = track.Length or 0
		Visualizer.paused = false
		btnPause.Text = "Pause"
		refreshBadge()
	end

	local function togglePause()
		if not Visualizer.track then return end
		Visualizer.paused = not Visualizer.paused
		if Visualizer.paused then
			Visualizer.pausedTime = Visualizer.track.TimePosition
			pcall(function() Visualizer.track:AdjustSpeed(0) end)
			pcall(function() Visualizer.track:Pause() end)
			btnPause.Text = "Resume"
		else
			pcall(function() Visualizer.track:Play() end)
			pcall(function() Visualizer.track.TimePosition = Visualizer.pausedTime end)
			pcall(function() Visualizer.track:AdjustSpeed(Visualizer.speed) end)
			btnPause.Text = "Pause"
		end
	end

	RunService.RenderStepped:Connect(function()
		if not Visualizer.track then return end
		if Visualizer.paused then
			pcall(function() Visualizer.track.TimePosition = Visualizer.pausedTime end)
		end
		local t   = Visualizer.track.TimePosition or 0
		local dur = Visualizer.track.Length or Visualizer.duration or 0
		Visualizer.duration = dur
		local frac = dur > 0 and (t / dur) or 0
		if frac > 1 then frac = 1 end
		scrubFill.Size = UDim2.new(frac, 0, 1, 0)
		timeLbl.Text = string.format("%.3fs / %.3fs", t, dur)
	end)

	local dragging = false
	scrubBg.InputBegan:Connect(function(input)
		if input.UserInputType == Enum.UserInputType.MouseButton1
			or input.UserInputType == Enum.UserInputType.Touch then
			dragging = true
			if Visualizer.track then
				local width = scrubBg.AbsoluteSize.X
				if width > 0 then
					local f = math.clamp((input.Position.X - scrubBg.AbsolutePosition.X) / width, 0, 1)
					Visualizer.pausedTime = f * (Visualizer.track.Length or 0)
					pcall(function() Visualizer.track.TimePosition = Visualizer.pausedTime end)
				end
			end
		end
	end)
	scrubBg.InputChanged:Connect(function(input)
		if not dragging or not Visualizer.track then return end
		if input.UserInputType ~= Enum.UserInputType.MouseMovement
			and input.UserInputType ~= Enum.UserInputType.Touch then return end
		local width = scrubBg.AbsoluteSize.X
		if width > 0 then
			local f = math.clamp((input.Position.X - scrubBg.AbsolutePosition.X) / width, 0, 1)
			Visualizer.pausedTime = f * (Visualizer.track.Length or 0)
			pcall(function() Visualizer.track.TimePosition = Visualizer.pausedTime end)
		end
	end)
	scrubBg.InputEnded:Connect(function(input)
		if input.UserInputType == Enum.UserInputType.MouseButton1
			or input.UserInputType == Enum.UserInputType.Touch then
			dragging = false
		end
	end)

	btnPlay.MouseButton1Click:Connect(function()
		if Visualizer.currentId then play(Visualizer.currentId) end
	end)
	btnPause.MouseButton1Click:Connect(togglePause)
	btnStop.MouseButton1Click:Connect(stop)

	local SPEEDS = { 0.1, 0.25, 0.5, 1, 1.5, 2 }
	local si = 4
	btnSpeed.MouseButton1Click:Connect(function()
		si = si % #SPEEDS + 1
		Visualizer.speed = SPEEDS[si]
		btnSpeed.Text = "Speed " .. tostring(Visualizer.speed) .. "x"
		if Visualizer.track and not Visualizer.paused then
			pcall(function() Visualizer.track:AdjustSpeed(Visualizer.speed) end)
		end
	end)

	btnLoop.MouseButton1Click:Connect(function()
		Visualizer.looping = not Visualizer.looping
		btnLoop.Text = Visualizer.looping and "Loop ON" or "Loop OFF"
		if Visualizer.track then
			pcall(function() Visualizer.track.Looped = Visualizer.looping end)
		end
	end)

	btnMove.MouseButton1Click:Connect(function()
		local next = not MoveLock.enabled
		setMoveLock(next)
		btnMove.Text = next and "Move ON" or "Move OFF"
		btnMove.BackgroundColor3 = next and Color3.fromRGB(60, 130, 60) or Color3.fromRGB(90, 60, 60)
	end)

	btnClear.MouseButton1Click:Connect(function()
		timings = {}
		saveTimings()
		refreshBadge()
		refreshDots()
	end)

	UserInputService.InputBegan:Connect(function(input, gp)
		if gp or input.KeyCode ~= Enum.KeyCode.Space then return end
		if not frame.Visible then return end
		togglePause()
	end)

	LP.CharacterAdded:Connect(function()
		task.wait(1)
		buildRig()
		if Visualizer.track and not Visualizer.paused then
			pcall(function() Visualizer.track:Stop() end)
			Visualizer.track = nil
		end
	end)

	frame:GetPropertyChangedSignal("Visible"):Connect(function()
		restore.Visible = not frame.Visible
	end)

	Visualizer.frame = frame
	Visualizer.idBar = idBar
	Visualizer.playFn = play
	Visualizer.btnMove = btnMove
end

local logRange = 120
local enemyHooked = {}
local selfHooked = {}

local function hookEnemy(hum)
	if not hum or not hum:IsA("Humanoid") or enemyHooked[hum] then return end
	local animator = hum:FindFirstChildOfClass("Animator")
	if not animator then
		task.spawn(function()
			local waited = 0
			while hum.Parent and not animator and waited < 5 do
				task.wait(0.1); waited += 0.1
				animator = hum:FindFirstChildOfClass("Animator")
			end
			if animator then hookEnemy(hum) end
		end)
		return
	end
	enemyHooked[hum] = true
	local model = hum.Parent

	animator.AnimationPlayed:Connect(function(track)
		local anim = track.Animation
		if not anim then return end
		local id = anim.AnimationId
		if not id or id == "" then return end
		if isLoco(id, anim.Name) then return end
		if model == LP.Character then return end

		local myRoot = LP.Character and LP.Character:FindFirstChild("HumanoidRootPart")
		local theirRoot = model:FindFirstChild("HumanoidRootPart")
		if not myRoot or not theirRoot then return end

		local d = (myRoot.Position - theirRoot.Position).Magnitude
		if d <= logRange then
			task.spawn(function()
				enemyLog.addRow(id, anim.Name, model.Name, math.floor(d))
			end)
		end
	end)
end

local function hookSelf(hum)
	if not hum or not hum:IsA("Humanoid") or selfHooked[hum] then return end
	local animator = hum:FindFirstChildOfClass("Animator")
	if not animator then
		task.spawn(function()
			local waited = 0
			while hum.Parent and not animator and waited < 5 do
				task.wait(0.1); waited += 0.1
				animator = hum:FindFirstChildOfClass("Animator")
			end
			if animator then hookSelf(hum) end
		end)
		return
	end
	selfHooked[hum] = true

	animator.AnimationPlayed:Connect(function(track)
		local anim = track.Animation
		if not anim then return end
		local id = anim.AnimationId
		if not id or id == "" then return end
		if isLoco(id, anim.Name) then return end
		task.spawn(function()
			selfLog.addRow(id, anim.Name, "me", 0)
		end)
	end)
end

local function scanEnemies()
	local myChar = LP.Character
	for _, o in ipairs(workspace:GetDescendants()) do
		if o:IsA("Humanoid") and not (o.Parent and myChar and o.Parent == myChar) then
			hookEnemy(o)
		end
	end
end

local function bindSelf()
	local char = LP.Character
	if not char then return end
	local hum = char:FindFirstChildOfClass("Humanoid")
	if hum then hookSelf(hum) end
	char.ChildAdded:Connect(function(c)
		if c:IsA("Humanoid") then hookSelf(c) end
	end)
end
